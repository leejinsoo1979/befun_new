import { chromium } from 'playwright-core';
const OUT = '/Users/leejinsoo/Downloads/befun_new-main/docs/handoffs/claude-1-ui-assets/r3';
const browser = await chromium.launch({ executablePath: '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome', headless: true, args: ['--use-angle=swiftshader', '--enable-unsafe-swiftshader'] });
const errs = [];
const setRange = (loc, v) => loc.evaluate((el, v) => { const s = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value').set; s.call(el, String(v)); el.dispatchEvent(new Event('input', { bubbles: true })); }, v);
const labels = (p) => p.evaluate(() => [...document.querySelectorAll('.cfg-label')].filter(e => e.offsetParent).map(e => e.textContent.trim()));
const overflow = (p) => p.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
for (const url of ['/configurator', '/furniture/sideboard-terra']) {
  const tag = url === '/configurator' ? 'configurator' : 'sideboard';
  // desktop
  const p = await browser.newPage({ viewport: { width: 1440, height: 1000 } });
  p.on('pageerror', e => errs.push(`${tag}: ${e.message}`)); p.on('console', m => { if (m.type() === 'error') errs.push(`${tag} console: ${m.text().slice(0, 160)}`); });
  await p.goto('http://localhost:3003' + url, { waitUntil: 'networkidle', timeout: 180000 }); await p.waitForTimeout(6000);
  console.log(tag, 'desktop labels', JSON.stringify(await labels(p)));
  const H = p.getByRole('slider', { name: 'Height' }).locator('visible=true').first();
  const pillH = () => H.getAttribute('aria-valuetext');
  await p.getByRole('button', { name: 'Grid' }).locator('visible=true').first().click();
  await setRange(p.getByRole('slider', { name: 'Width' }).locator('visible=true').first(), 110);
  await setRange(H, 2);
  await p.getByRole('button', { name: '32cm' }).locator('visible=true').first().click();
  await p.waitForTimeout(500);
  const std = await pillH();
  const legsBtn = p.getByRole('button', { name: 'Legs' }).locator('visible=true').first();
  console.log(tag, 'legs enabled', await legsBtn.isEnabled());
  await legsBtn.click(); await p.waitForTimeout(600);
  const legs = await pillH();
  const spec = tag === 'sideboard' ? await p.evaluate(() => [...document.querySelectorAll('dd')].map(e => e.textContent).find(t => /×.*×/.test(t))) : null;
  console.log(tag, 'R3-UI-01 Standard', std, '→ Legs', legs, spec ? `spec ${spec}` : '');
  await p.getByRole('button', { name: '치수' }).first().click().catch(() => {});
  await p.waitForTimeout(1200);
  await p.screenshot({ path: `${OUT}/${tag}-desktop-legs-dims.png` });
  // keyboard focus
  await H.focus(); await p.keyboard.press('ArrowRight'); await p.waitForTimeout(300);
  console.log(tag, 'kbd rows→3', await pillH(), 'focus kept', await H.evaluate(e => e === document.activeElement));
  // R3-UI-02 depth 24
  await p.getByRole('button', { name: '24cm' }).locator('visible=true').first().click(); await p.waitForTimeout(500);
  console.log(tag, 'R3-UI-02 depth24 legs disabled', !(await legsBtn.isEnabled()), 'Standard pressed', await p.getByRole('button', { name: 'Standard' }).locator('visible=true').first().getAttribute('aria-pressed'), 'height', await pillH());
  console.log(tag, 'desktop overflow', await overflow(p));
  await p.screenshot({ path: `${OUT}/${tag}-desktop.png`, fullPage: false });
  await p.close();
  // mobile
  const m = await browser.newPage({ viewport: { width: 390, height: 844 }, hasTouch: true, isMobile: true });
  m.on('pageerror', e => errs.push(`${tag} m: ${e.message}`));
  await m.goto('http://localhost:3003' + url, { waitUntil: 'networkidle', timeout: 180000 }); await m.waitForTimeout(6000);
  const tabs = m.getByRole('tab');
  if (await tabs.count()) {
    for (const name of ['스타일', '크기', '색상', '옵션']) {
      const t = m.getByRole('tab', { name }); if (!(await t.count())) continue;
      await t.first().click(); await m.waitForTimeout(400);
      console.log(tag, 'mobile tab', name, JSON.stringify(await labels(m)), 'overflow', await overflow(m));
      await m.screenshot({ path: `${OUT}/${tag}-390-${name}.png` });
    }
  } else {
    console.log(tag, 'mobile labels', JSON.stringify(await labels(m)), 'overflow', await overflow(m));
    await m.screenshot({ path: `${OUT}/${tag}-390.png`, fullPage: true });
  }
  await m.close();
}
console.log('errors', JSON.stringify(errs.slice(0, 8)));
await browser.close();
