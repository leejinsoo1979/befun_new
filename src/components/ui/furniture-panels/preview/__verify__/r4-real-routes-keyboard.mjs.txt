const browser = await chromium.launch({ executablePath: '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome', headless: true, args: ['--use-angle=swiftshader', '--enable-unsafe-swiftshader'] });

const errs = []; const rows = [];
const rec = (id, step, expect, actual, ok) => { rows.push({ id, step, expect, actual, ok }); console.log(ok ? 'PASS' : 'FAIL', id, '|', step, '|', expect, '|', actual); };
async function open(url) {
  const p = await browser.newPage({ viewport: { width: W, height: H }, hasTouch: W < 500, isMobile: W < 500 });
  p.on('pageerror', e => errs.push(`${url}: ${e.message}`));
  await p.goto(BASE + url, { waitUntil: 'domcontentloaded', timeout: 180000 });
  await p.waitForSelector('[role=tablist]', { timeout: 120000 }); await p.waitForTimeout(2000);
  return p;
}
const S = (p) => ({
  tab: () => p.evaluate(() => document.querySelector('[role=tab][aria-selected=true]')?.textContent),
  section: () => p.evaluate(() => document.querySelector('[aria-label="구획 선택"] [aria-checked=true]')?.textContent ?? null),
  checked: (label) => p.evaluate((l) => { const g = [...document.querySelectorAll('[role=radiogroup]')].find(x => x.getAttribute('aria-label') === l); if (!g) return null; const r = [...g.querySelectorAll('[role=radio]')].find(r => r.getAttribute('aria-checked') === 'true'); return r ? (r.textContent || r.getAttribute('aria-label')) : '(none)'; }, label),
  has: (id) => p.evaluate((id) => !!document.querySelector(`[data-control-id="${id}"]`), id),
  focusLabel: () => p.evaluate(() => { const a = document.activeElement; return a ? (a.getAttribute('aria-label') || a.textContent || a.tagName).trim().slice(0, 30) : null; }),
  slider: (name) => p.getByRole('slider', { name, exact: true }).first().evaluate(e => ({ v: e.value, t: e.getAttribute('aria-valuetext'), step: e.getAttribute('step'), min: e.min, max: e.max })),
});
const kbTab = async (p, name) => { const t = p.getByRole('tab', { name }); await t.focus(); await p.keyboard.press('Enter'); };
const radio = (p, group, name) => p.getByRole('radiogroup', { name: group }).getByRole('radio', { name, exact: true });
const pressRadio = async (p, group, name) => { const r = radio(p, group, name); await r.focus(); await p.keyboard.press('Space'); await p.waitForTimeout(250); };
