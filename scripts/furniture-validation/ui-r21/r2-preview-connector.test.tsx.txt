import { describe, it, expect, vi } from 'vitest';
import { render, screen, fireEvent, within } from '@testing-library/react';
import { ConnectorPanelPreview } from '@/components/ui/furniture-panels/preview/ConnectorPanelPreview';
import { OVVO_CONNECTORS } from '@/lib/optimizer/connectors/ovvo';
import { CLIP_SPEC_PENDING } from '@/types/connectors';

const order = (el: HTMLElement) => [...el.querySelectorAll('[data-control-id]')].map(e => e.getAttribute('data-control-id'));
const group = (name: string) => screen.getByRole('radiogroup', { name });
const pick = (g: string, o: string) => fireEvent.click(within(group(g)).getByRole('radio', { name: o }));

describe('접합 방식 미리보기 (OVVO / Tylko 클립)', () => {
  it('OVVO: 기존 옵션 뒤에 방식·모델·가공 상태, 클립 모델은 숨김', () => {
    const { container } = render(<ConnectorPanelPreview />);
    expect(order(container)).toEqual(['style', 'density', 'width', 'height', 'depth', 'backPanel', 'finish', 'color', 'connectorSystem', 'connectorBody', 'connectorBack', 'machiningStatus']);
    expect(within(group('접합 방식')).getByRole('radio', { name: 'OVVO 방식' })).toHaveAttribute('aria-checked', 'true');
    expect(screen.getByText('V-1230R 가공 생성 대상(옵티마이저 기준)')).toBeTruthy();
  });
  it('OVVO 모델 활성 여부는 카탈로그 autoApplicable 그대로', () => {
    render(<ConnectorPanelPreview />);
    for (const c of OVVO_CONNECTORS) {
      const r = within(group('몸통 모델')).getByRole('radio', { name: c.model });
      expect(r.hasAttribute('disabled')).toBe(!c.autoApplicable);
    }
  });
  it('Tylko 클립 전환: OVVO 모델 숨김, 클립 모델 비활성+사유, 가공 미확정 문구', () => {
    const { container } = render(<ConnectorPanelPreview />);
    pick('접합 방식', 'Tylko 클립 방식');
    expect(order(container).slice(-3)).toEqual(['connectorSystem', 'clipModel', 'machiningStatus']);
    const clip = group('클립 모델');
    expect(within(clip).getAllByRole('radio').every(r => r.hasAttribute('disabled') && r.tabIndex === -1)).toBe(true);
    expect(within(clip).getAllByRole('radio').every(r => r.getAttribute('aria-checked') === 'false')).toBe(true);
    expect(screen.getByText('검증된 클립 모델 없음 — 품번·치수 미확정')).toBeTruthy();
    expect(screen.getByText(`가공 미확정: ${CLIP_SPEC_PENDING}`)).toBeTruthy();
  });
  it('방식 전환은 스타일·치수·재질·이전 OVVO 모델 값을 지우지 않는다', () => {
    const onValues = vi.fn();
    render(<ConnectorPanelPreview onValues={onValues} />);
    pick('스타일', 'Grid'); pick('깊이', '40cm'); pick('마감', '도장');
    fireEvent.change(screen.getByRole('slider', { name: '너비' }), { target: { value: '180' } });
    pick('백패널 모델', 'PD20');
    pick('접합 방식', 'Tylko 클립 방식');
    const during = onValues.mock.lastCall![0];
    expect(during).toMatchObject({ connectorSystem: 'tylko-clip', style: 'grid', depth: '40', finish: 'paint', width: 180, connectorBody: 'V-1230R', connectorBack: 'PD20', color: 'S_White' });
    pick('색상', '세이지 그린'); // 클립 방식 중 다른 옵션 변경
    pick('접합 방식', 'OVVO 방식');
    expect(onValues.mock.lastCall![0]).toMatchObject({ connectorSystem: 'ovvo', style: 'grid', depth: '40', finish: 'paint', width: 180, color: 'S_Sage Green', connectorBody: 'V-1230R', connectorBack: 'PD20' });
    expect(within(group('몸통 모델')).getByRole('radio', { name: 'V-1230R' })).toHaveAttribute('aria-checked', 'true');
    expect(within(group('백패널 모델')).getByRole('radio', { name: 'PD20' })).toHaveAttribute('aria-checked', 'true');
    expect(screen.getByText('V-1230R · PD20 가공 생성 대상(옵티마이저 기준)')).toBeTruthy();
  });
  it('방식 전환 intent는 connectorSystem set 하나뿐', () => {
    const onValues = vi.fn();
    render(<ConnectorPanelPreview onValues={onValues} />);
    pick('접합 방식', 'Tylko 클립 방식');
    expect(onValues).toHaveBeenCalledTimes(1);
  });
  it('키보드: 방식 라디오 화살표로 전환, 모델 없음 선택 시 가공 없음', () => {
    render(<ConnectorPanelPreview />);
    const ovvo = within(group('접합 방식')).getByRole('radio', { name: 'OVVO 방식' });
    expect(ovvo.tabIndex).toBe(0);
    fireEvent.keyDown(ovvo, { key: 'ArrowRight' });
    expect(within(group('접합 방식')).getByRole('radio', { name: 'Tylko 클립 방식' })).toHaveAttribute('aria-checked', 'true');
    fireEvent.keyDown(within(group('접합 방식')).getByRole('radio', { name: 'Tylko 클립 방식' }), { key: 'ArrowLeft' });
    pick('몸통 모델', '없음');
    expect(screen.getByText('모델 미선택 — 접합 가공 없음')).toBeTruthy();
  });
});
